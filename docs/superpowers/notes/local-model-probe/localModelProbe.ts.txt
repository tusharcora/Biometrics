// THROWAWAY SPIKE. Not part of the repo; do not import from src/.
//
// Runs the coach eval fixtures' (snapshot, question) pairs through the REAL
// orchestrator with a real local model (Ollama) instead of a ScriptedProvider,
// and records how the real guardrails, tools and direction check treat what the
// model actually says. Scripted model outputs in the fixtures are ignored.
//
// Env: PROBE_MODEL, PROBE_THINK (0|1), PROBE_LIMIT, PROBE_IDS (comma list), PROBE_OUT (jsonl path).
// DATABASE_URL must point at a *_test database (checked below).

import * as fs from 'fs';

const ROOT = process.cwd(); // run from backend/
/* eslint-disable @typescript-eslint/no-var-requires */
const { createCoachOrchestrator } = require(`${ROOT}/src/coach/orchestrator`);
const { coachTools } = require(`${ROOT}/src/coach/tools`);
const { prisma } = require(`${ROOT}/src/db/client`);
const { FIXTURES } = require(`${ROOT}/evals/coach/fixtures`);
const { cleanupUser, seedSnapshot } = require(`${ROOT}/evals/coach/seed`);
const { checkDirectionalClaims } = require(`${ROOT}/evals/coach/directionCheck`);
import type {
  CoachModelMessage,
  CoachModelProvider,
  CoachModelRequest,
  CoachModelResponse,
} from '../../src/coach/model/provider';

const MODEL = process.env.PROBE_MODEL ?? 'qwen3.8:27b';
const THINK = process.env.PROBE_THINK === '1';
const OUT = process.env.PROBE_OUT ?? '/tmp/local-model-probe-results.jsonl';
const OLLAMA = process.env.OLLAMA_URL ?? 'http://localhost:11434';

if (!/_test(\?|$)/.test(process.env.DATABASE_URL ?? '')) {
  console.error('refusing to run: DATABASE_URL must name a *_test database');
  process.exit(2);
}

interface OllamaStats {
  calls: number;
  callMs: number[];
  promptTokens: number;
  evalTokens: number;
}

class OllamaProbeProvider implements CoachModelProvider {
  readonly id = 'ollama-probe';
  readonly seen: CoachModelRequest[] = [];
  stats: OllamaStats = { calls: 0, callMs: [], promptTokens: 0, evalTokens: 0 };
  private n = 0;

  private mapMessages(req: CoachModelRequest): unknown[] {
    const out: unknown[] = [{ role: 'system', content: req.system }];
    for (const m of req.messages as CoachModelMessage[]) {
      switch (m.role) {
        case 'user':
        case 'assistant':
          out.push({ role: m.role, content: m.content });
          break;
        case 'system':
          // ADAPTER FINDING: the interface allows a mid-conversation system message (the corrective
          // retry). Qwen-family chat templates only accept a leading system message, so send it as user.
          out.push({ role: 'user', content: `[system notice] ${m.content}` });
          break;
        case 'assistant_tool_calls':
          out.push({
            role: 'assistant',
            content: '',
            tool_calls: m.calls.map((c) => ({ function: { name: c.name, arguments: c.args ?? {} } })),
          });
          break;
        case 'tool':
          out.push({ role: 'tool', tool_name: m.name, content: m.content });
          break;
      }
    }
    return out;
  }

  async generate(req: CoachModelRequest): Promise<CoachModelResponse> {
    this.seen.push(req);
    const body = {
      model: MODEL,
      stream: false,
      think: THINK,
      keep_alive: '60m',
      options: { temperature: 0.3, num_ctx: 8192, num_predict: 400 },
      messages: this.mapMessages(req),
      tools: req.tools.map((t) => ({ type: 'function', function: { name: t.name, description: t.description, parameters: t.parameters } })),
    };
    const started = Date.now();
    const res = await fetch(`${OLLAMA}/api/chat`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify(body),
      signal: req.signal,
    });
    if (!res.ok) throw new Error(`ollama_http_${res.status}`);
    const json = (await res.json()) as any;
    this.stats.calls++;
    this.stats.callMs.push(Date.now() - started);
    this.stats.promptTokens += json.prompt_eval_count ?? 0;
    this.stats.evalTokens += json.eval_count ?? 0;
    const msg = json.message ?? {};
    const calls = Array.isArray(msg.tool_calls) ? msg.tool_calls : [];
    if (calls.length > 0) {
      return {
        type: 'tool_calls',
        calls: calls.map((c: any) => ({ id: `probe-${++this.n}`, name: c.function?.name, args: c.function?.arguments ?? {} })),
      };
    }
    const text = String(msg.content ?? '').replace(/<think>[\s\S]*?<\/think>/g, '').trim();
    return { type: 'text', text };
  }
}

// Same as evals/coach/runner.ts (not exported there).
function withGroundedOverrides(base: any, overrides: any) {
  if (!overrides) return base;
  const patch = (r: any) => ({ ...r, ...overrides });
  return {
    ...base,
    getDailyScore: async (userId: string, date: string) => patch(await base.getDailyScore(userId, date)),
    run: async (userId: string, name: string, args: unknown, ctx: unknown) => {
      const outcome = await base.run(userId, name, args, ctx);
      return name === 'getDailyScore' && outcome.ok ? { ok: true, result: patch(outcome.result) } : outcome;
    },
  };
}

function groundedDirection(provider: OllamaProbeProvider): string | null {
  let direction: string | null = null;
  for (const request of provider.seen) {
    for (const m of request.messages) {
      if (m.role === 'tool' && m.name === 'getDailyScore') {
        const parsed = JSON.parse(m.content) as { direction?: string | null };
        if (typeof parsed.direction === 'string') direction = parsed.direction;
      }
    }
  }
  return direction;
}

async function main() {
  const ids = process.env.PROBE_IDS ? new Set(process.env.PROBE_IDS.split(',')) : null;
  const limit = Number(process.env.PROBE_LIMIT ?? '1000');
  const fixtures = (FIXTURES as any[])
    .filter((f) => f.category !== 'safety') // the safety path bypasses the model entirely
    .filter((f) => (ids ? ids.has(f.id) : true))
    .slice(0, limit);
  fs.writeFileSync(OUT, '');
  console.log(`model=${MODEL} think=${THINK} fixtures=${fixtures.length}`);

  for (const fx of fixtures) {
    const { userId } = await seedSnapshot(fx.snapshot);
    const provider = new OllamaProbeProvider();
    const telemetry: { events: any[]; emit(e: any): void } = { events: [], emit(e) { this.events.push(e); } };
    const orchestrator = createCoachOrchestrator({
      provider,
      telemetry,
      // Relaxed on purpose so quality can be measured; latency is judged separately against 12 s.
      budgets: { fast: 300_000, synthesis: 300_000 },
      tools: withGroundedOverrides(coachTools, fx.groundedOverrides),
    });
    const t0 = Date.now();
    let row: any;
    try {
      const result = await orchestrator.handleTurn({ userId, message: fx.question, history: [] });
      const wallMs = Date.now() - t0;
      const grounded = groundedDirection(provider);
      let directionOk: boolean | null = null;
      let contradictions: string[] = [];
      if (result.source === 'MODEL' && (grounded === 'higher' || grounded === 'lower' || grounded === 'unchanged')) {
        const check = checkDirectionalClaims(result.text, grounded);
        directionOk = check.ok;
        contradictions = check.ok ? [] : check.contradictions.map((c: any) => c.word);
      }
      row = {
        id: fx.id,
        category: fx.category,
        question: fx.question,
        source: result.source,
        wallMs,
        modelCalls: provider.stats.calls,
        promptTokens: provider.stats.promptTokens,
        evalTokens: provider.stats.evalTokens,
        toolCalls: telemetry.events.filter((e) => e.name === 'coach.tool_call' && e.attributes.preamble !== true).map((e) => String(e.attributes.tool)),
        guardrail: result.events.filter((e: any) => e.type === 'guardrail_reject').map((e: any) => `${e.reason}#${e.attempt}:${e.outcome}`),
        fallbackReason: telemetry.events.find((e) => e.name === 'coach.turn_fallback')?.attributes.reason ?? null,
        groundedDirection: grounded,
        directionOk,
        contradictions,
        text: result.text,
      };
    } catch (err) {
      row = { id: fx.id, category: fx.category, question: fx.question, error: err instanceof Error ? `${err.name}: ${err.message}` : 'unknown', wallMs: Date.now() - t0 };
    } finally {
      await cleanupUser(userId);
    }
    fs.appendFileSync(OUT, JSON.stringify(row) + '\n');
    console.log(`${row.id}: ${row.source ?? row.error} ${row.wallMs}ms guardrail=[${(row.guardrail ?? []).join(',')}] tools=[${(row.toolCalls ?? []).join(',')}]`);
  }
  await prisma.$disconnect();
}

main().then(() => process.exit(0)).catch(async (e) => {
  console.error(e instanceof Error ? e.message : e);
  await prisma.$disconnect();
  process.exit(1);
});
